\section{Recall}

\textbf{有界性：}\quad 若 $\{a_n\}$ 满足
$\lim\limits_{n\to+\infty}a_n=a$ 存在，那么 $\{a_n\}$ 有界。

\textbf{保号性：}\quad $\lim\limits_{n\to+\infty}a_n=a$，那么下面的性质成立：

\begin{itemize}
\item 若 $a>0$，则 $\exists N,\ \forall n>N,\ a_n>0$；
\item 若 $\exists N$，s.t. $\forall n>N,\ a_n\geq0$，则 $a\geq0$。
\end{itemize}

下给出证明。

\noindent 1) 由 $\lim\limits_{n\to+\infty}a_n=a$ 可知：
$\forall\epsilon>0$，$\exists N$，$\forall n>N$，有
$|a_n-a|<\epsilon$。若 $a>0$，取 $\epsilon_0=\dfrac12a$，
对于此 $\epsilon_0$，$\exists N$，$\forall n>N$ 有
\[
|a_n-a|<\frac12a=\epsilon_0
\ \Longrightarrow\
a_n-a>-\frac12a
\ \Longrightarrow\
a_n>\frac12a>0 .
\]

\noindent 2) $\forall\epsilon>0$，$\exists N$，$\forall n>N$，
$|a_n-a|<\epsilon$，可得
\[
\forall n>N,\quad a>a_n-\epsilon .
\]
再由 1) 得 $\forall n>M,\ a_n\geq0$
（这里 $M$ 就是 1) 中的 $N$，换一个字母以免与上式的 $N$ 混用）。
取 $n\geq\max\{M,N\}$，有 $a>a_n-\epsilon$ 且 $a_n\geq0$，故
\[
a>-\epsilon\ \Longrightarrow\ a\geq0 .
\]

即使 2) 中要求 $\exists N$，s.t. $\forall n>N,\ a_n>0$，
仍然只能得到 $a\geq0$ 而不是 $a>0$。
如 $a_n=\dfrac1n$，$\forall n$ 有 $a_n>0$，但
$\lim\limits_{n\to+\infty}a_n=0$。

\section{极限的唯一性}

\begin{thm}
若 $\lim\limits_{n\to+\infty}a_n$ 存在，
则极限值 $\lim\limits_{n\to+\infty}a_n$ 唯一。
\end{thm}

分析：反设如果存在两个极限
$\lim\limits_{n\to+\infty}a_n=a$ 和
$\lim\limits_{n\to+\infty}a_n=b$，我们目前只知道极限定义，
我们把这两个极限写出来，即
\begin{align*}
&\forall\epsilon>0,\ \exists N,\ \forall n>N,\ |a_n-a|<\epsilon,\\
&\forall\epsilon>0,\ \exists M,\ \forall n>M,\ |a_n-b|<\epsilon .
\end{align*}
如果对于 $\forall\epsilon>0$，总能取到同一个 $n$，s.t.
$|a_n-a|<\epsilon$ 且 $|a_n-b|<\epsilon$，
可得 $|a-b|<2\epsilon$，即 $a=b$。

\textbf{证明：}\quad 取 $\tilde N=\max\{M,N\}$，那么只要
$n>\tilde N$，有 $|a_n-a|<\epsilon$ 且 $|a_n-b|<\epsilon$，
可得 $|a-b|<2\epsilon$，即 $a=b$。

如果已得到下面的事实：
$\forall\varepsilon>0$，$\exists N$，$\forall n>N$，
$|c_n-A|<k\varepsilon$，其中 $k$ 是固定的、不依赖于
$\varepsilon$ 和 $N$ 的正数，则
$\lim\limits_{n\to\infty}c_n=A$。

证明：$\forall\varepsilon>0$，找 $M$，$\forall n>M$，有
$|c_n-A|<\varepsilon$。对于 $\varepsilon>0$，
考虑 $\dfrac1k\varepsilon>0$，将其记为 $\varepsilon'$，则
$\exists N_{\frac1k\varepsilon}$，$\forall n>N_{\frac1k\varepsilon}$，s.t.
\[
|c_n-A|<k\cdot\frac1k\varepsilon=\varepsilon .
\]

\section{极限的四则运算法则}

若 $\lim\limits_{n\to\infty}a_n=a$ 且
$\lim\limits_{n\to\infty}b_n=b$，则：

\begin{itemize}
\item $\lim\limits_{n\to\infty}(a_n\pm b_n)=a\pm b$，
$\lim\limits_{n\to\infty}(a_nb_n)=ab$；
\item if $b\neq0$：
$\lim\limits_{n\to\infty}\left(\dfrac{a_n}{b_n}\right)=\dfrac ab$。
\end{itemize}

\subsubsection*{加法证明}

设 $\lim\limits_{n\to\infty}a_n=a$，
$\lim\limits_{n\to\infty}b_n=b$。
$\forall\varepsilon>0$，$\exists N>0$，对 $\forall n>N$ s.t.
$|a_n-a|<\varepsilon$；$\exists M$ s.t. $\forall n>M$ s.t.
$|b_n-b|<\varepsilon$。

令 $\tilde N=\max\{M,N\}$，只要 $n>\tilde N$，有
$|a_n-a|<\varepsilon$ 且 $|b_n-b|<\varepsilon$。

令 $c_n=a_n+b_n$，那么
\[
|c_n-(a+b)|=|(a_n-a)+(b_n-b)|
\leq|a_n-a|+|b_n-b|<2\varepsilon .
\]
$\Longrightarrow\ \lim\limits_{n\to\infty}(a_n+b_n)=a+b$。

\subsubsection*{乘法证明}

$\lim\limits_{n\to\infty}a_n=a$ 且
$\lim\limits_{n\to\infty}b_n=b$，则
$\lim\limits_{n\to\infty}a_n\cdot b_n=a\cdot b$。

令 $c_n=a_n\cdot b_n$，则 $\{c_n\}$ 是一个数列，
目标 $\lim\limits_{n\to\infty}c_n=a\cdot b$。

\textbf{证明}\quad
$\forall\varepsilon>0$，$\exists N_1$，$\forall n>N_1$，
$|a_n-a|<\varepsilon$；$\forall\varepsilon>0$，$\exists N_2$，
$\forall n>N_2$，$|b_n-b|<\varepsilon$。

考虑 $|a_nb_n-ab|$：
\[
|a_nb_n-ab|
=|a_nb_n-a_nb+a_nb-ab|
=|a_n(b_n-b)+b(a_n-a)|
\leq|a_n||b_n-b|+|b||a_n-a| .
\]
又 $\lim\limits_{n\to\infty}a_n=a$，$\{a_n\}$ 有界，
$\exists M>0$，s.t. $\forall n,\ |a_n|\leq M$。

取 $\tilde N=\max\{N_1,N_2\}$，只要 $n>\tilde N$，有：
\[
|a_nb_n-ab|\leq M\cdot\varepsilon+|b|\cdot\varepsilon
=(M+|b|)\varepsilon
\ \Longrightarrow\
\lim_{n\to\infty}a_nb_n=a\cdot b .
\]

\subsubsection*{除法证明}

$\lim\limits_{n\to\infty}a_n=a$，
$\lim\limits_{n\to\infty}b_n=b\neq0$，目标
$\lim\limits_{n\to\infty}\dfrac{a_n}{b_n}=\dfrac ab$。

$\forall\varepsilon>0$，$\exists N$，$\forall n>N$，
$|a_n-a|<\varepsilon$ 且 $|b_n-b|<\varepsilon$。考虑：
\[
\left|\frac{a_n}{b_n}-\frac ab\right|
=\left|\frac{a_nb-ab_n}{b_nb}\right|
=\frac{|a_nb-ab+ab-ab_n|}{|b_nb|}
\leq\frac{|b||a_n-a|+|a||b_n-b|}{|b_nb|} .
\]
不妨设 $b>0$，由保号性 $\exists M$，$\forall n>M$，
$b_n>\dfrac12b$（$\varepsilon$ 取 $\dfrac12b$）；
若 $b<0$，对 $-b_n$ 用同样的论证。

$\forall n>\max\{N,M\}$，有：
\[
\left|\frac{a_n}{b_n}-\frac ab\right|
\leq\frac{|b||a_n-a|+|a||b_n-b|}{|b_nb|}
<\frac{(|b|+|a|)\varepsilon}{\frac12b^2}
=\frac{2(|b|+|a|)}{b^2}\,\varepsilon .
\]

\section{夹迫原理（sandwich theorem）}

\begin{thm}
若 $\exists N$ s.t. $\forall n>N$，有
$a_n\leq c_n\leq b_n$ 且
\[
\lim_{n\to\infty}a_n=\lim_{n\to\infty}b_n=A,
\]
则
\[
\lim_{n\to\infty}c_n=A .
\]
\end{thm}

\textbf{证明：}\quad $\forall\varepsilon>0$，$\exists N_a$，
$\forall n>N_a$，有 $|a_n-A|<\varepsilon$；
$\exists N_b$，$\forall n>N_b$，有 $|b_n-A|<\varepsilon$
且 $a_n\leq c_n\leq b_n$（$n>N$）。

取 $M=\max\{N,N_a,N_b\}$。由 $|a_n-A|<\varepsilon$ 知
$A-\varepsilon<a_n<A+\varepsilon$；由 $|b_n-A|<\varepsilon$ 知
$A-\varepsilon<b_n<A+\varepsilon$。有
\[
A-\varepsilon<a_n\leq c_n\leq b_n<A+\varepsilon
\ \Longrightarrow\
A-\varepsilon<c_n<A+\varepsilon,
\]
即
\[
|c_n-A|<\varepsilon
\qquad\therefore\quad
\lim_{n\to\infty}c_n=A .
\]
